// ─── CAMBIO 1: imagen con altura fija en px (no h-32/h-40 de Tailwind)
// Esto hace que el ghost container siempre tenga dimensiones estables
// antes y después de que la imagen cargue, evitando saltos de layout.
function AnnouncementCard({ ann }: { ann: Announcement }) {
  return (
    <div className="flex flex-col gap-2 lg:gap-3">
      <div className="flex items-center gap-2">
        <span
          className="text-[9px] font-bold tracking-[0.18em] px-2 py-0.5 rounded-full border"
          style={{
            color: ann.accent,
            borderColor: `${ann.accent}40`,
            background: `${ann.accent}14`,
          }}
        >
          {ann.label}
        </span>
        <span className="text-[9px] uppercase tracking-widest" style={{ color: `${ann.accent}50` }}>
          {ann.type}
        </span>
      </div>

      {ann.type === "image" && ann.imageUrl && (
        // Altura fija en px (no clases responsivas) → el ghost la mide igual que la visible
        <div className="relative w-full rounded-xl overflow-hidden" style={{ height: "128px" }}>
          <Image
            src={ann.imageUrl}
            alt={ann.title}
            fill
            className="object-cover"
            sizes="(max-width: 640px) 100vw, 50vw"
          />
        </div>
      )}

      {ann.type === "video" && ann.videoUrl && (
        <div className="relative w-full rounded-xl overflow-hidden" style={{ aspectRatio: "16/9" }}>
          <iframe
            src={ann.videoUrl}
            className="absolute inset-0 w-full h-full"
            allow="accelerometer; autoplay; clipboard-write; encrypted-media; gyroscope; picture-in-picture"
            allowFullScreen
          />
        </div>
      )}

      <p className="text-sm lg:text-base font-semibold leading-snug text-sky-100/85">{ann.title}</p>
      <p className="text-[11px] lg:text-sm leading-relaxed text-sky-300/50">{ann.body}</p>
    </div>
  );
}

// ─── CAMBIO 2: panel extraído como componente propio con ghost layer
// El ghost renderiza todas las tarjetas invisibles (visibility:hidden, no display:none)
// para que el DOM calcule su altura real incluyendo imágenes.
// onLoad en las imágenes del ghost dispara una re-medición.
// La altura máxima se fija en el contenedor visible → nunca hay saltos.
function AnnouncementPanel({ activeAnns, annLabel, annAria }: { activeAnns: Announcement[]; annLabel: string; annAria: string }) {
  const [index, setIndex]   = useState(0);
  const [panelH, setPanelH] = useState<number | null>(null);
  const ghostRef            = useRef<HTMLDivElement>(null);
  const autoRef             = useRef<ReturnType<typeof setInterval> | null>(null);

  const measure = useCallback(() => {
    if (!ghostRef.current) return;
    const cards = ghostRef.current.querySelectorAll<HTMLElement>("[data-ghost-card]");
    let max = 0;
    cards.forEach((el) => { if (el.offsetHeight > max) max = el.offsetHeight; });
    if (max > 0) setPanelH(max);
  }, []);

  useEffect(() => {
    measure();
    const t1 = setTimeout(measure, 150);
    const t2 = setTimeout(measure, 600);

    const observer = new ResizeObserver(measure);
    if (ghostRef.current) {
      const cards = ghostRef.current.querySelectorAll<HTMLElement>("[data-ghost-card]");
      cards.forEach((card) => observer.observe(card));
    }

    window.addEventListener("resize", measure);
    return () => {
      clearTimeout(t1);
      clearTimeout(t2);
      window.removeEventListener("resize", measure);
      observer.disconnect();
    };
  }, [measure, activeAnns]);

  useEffect(() => {
    if (activeAnns.length <= 1) return;
    autoRef.current = setInterval(() => setIndex((i) => (i + 1) % activeAnns.length), 5000);
    return () => { if (autoRef.current) clearInterval(autoRef.current); };
  }, [activeAnns.length]);

  const bump = (dir: 1 | -1) => {
    if (autoRef.current) clearInterval(autoRef.current);
    setIndex((i) => (i + dir + activeAnns.length) % activeAnns.length);
  };

  return (
    <div className="flex-1 rounded-2xl p-4 sm:p-5 lg:p-6 flex flex-col gap-3 lg:gap-4 border border-sky-300/20 bg-sky-300/5">
      {/* Header */}
      <div className="flex items-center justify-between">
        <span className="text-[9px] lg:text-[10px] font-bold tracking-[0.22em] uppercase text-sky-300/40">
          {annLabel}
        </span>
        {activeAnns.length > 1 && (
          <div className="flex items-center gap-1.5">
            <button onClick={() => bump(-1)} className="p-0.5 rounded-lg text-sky-300/40 hover:text-sky-300/80 transition-colors">
              <ChevronLeft className="h-3.5 w-3.5 lg:h-4 lg:w-4" />
            </button>
            <span className="text-[9px] lg:text-[10px] tabular-nums text-sky-300/30">
              {index + 1}/{activeAnns.length}
            </span>
            <button onClick={() => bump(1)} className="p-0.5 rounded-lg text-sky-300/40 hover:text-sky-300/80 transition-colors">
              <ChevronRight className="h-3.5 w-3.5 lg:h-4 lg:w-4" />
            </button>
          </div>
        )}
      </div>

      {/* Ghost layer: in normal flow (so width matches panel), zero height so it
          doesn't affect layout. visibility:hidden keeps images loading & measurable. */}
      <div
        ref={ghostRef}
        className="pointer-events-none"
        style={{ visibility: "hidden", height: 0, overflow: "hidden" }}
        aria-hidden
      >
        {activeAnns.map((ann) => (
          <div key={ann.id} data-ghost-card className="pb-3">
            {ann.type === "image" && ann.imageUrl ? (
              <div className="flex flex-col gap-2">
                <div className="relative w-full rounded-xl overflow-hidden" style={{ height: "128px" }}>
                  <Image
                    src={ann.imageUrl}
                    alt=""
                    fill
                    className="object-cover"
                    sizes="(max-width: 640px) 100vw, 50vw"
                    onLoad={measure}
                  />
                </div>
                <p className="text-sm font-semibold">{ann.title}</p>
                <p className="text-[11px]">{ann.body}</p>
              </div>
            ) : (
              <AnnouncementCard ann={ann} />
            )}
          </div>
        ))}
      </div>

      {/* Contenedor visible con altura bloqueada */}
      <div
        className="relative"
        style={{ minHeight: panelH ? `${panelH}px` : undefined }}
      >
        <AnimatePresence mode="wait">
          <motion.div
            key={activeAnns[index].id}
            initial={{ opacity: 0, x: 8 }}
            animate={{ opacity: 1, x: 0 }}
            exit={{ opacity: 0, x: -8 }}
            transition={{ duration: 0.2 }}
            className="relative"
          >
            <AnnouncementCard ann={activeAnns[index]} />
          </motion.div>
        </AnimatePresence>
      </div>

      {/* Dots */}
      {activeAnns.length > 1 && (
        <div className="flex gap-1.5 justify-center mt-1">
          {activeAnns.map((_, i) => (
            <button
              key={i}
              onClick={() => { if (autoRef.current) clearInterval(autoRef.current); setIndex(i); }}
              className="h-1 rounded-full transition-all duration-300"
              style={{
                width: i === index ? "16px" : "5px",
                backgroundColor: i === index ? activeAnns[index].accent : "rgba(127,192,223,0.3)",
              }}
              aria-label={`${annAria} ${i + 1}`}
            />
          ))}
        </div>
      )}
    </div>
  );
}

// ─── Todo lo de abajo es idéntico al original ─────────────────────────────────
export default function ClientHome({ lang, home }: Props) {

  const activeAnns = (home.items as Announcement[]).filter((a) => a.active);

    setIsMounted(true);
    // Desktop only: lock body scroll since layout is fixed height
    if (window.innerWidth >= 768) {
      document.body.style.overflow = "hidden";
    }
    return () => {
      document.body.style.overflow = "";
    };
  }, []);

  if (!isMounted) {
    return (
      <main className="h-screen flex items-center justify-center bg-[#0a0c15]">
        <span className="text-xs text-sky-300/40 tracking-widest">{home.loading}</span>
      </main>

  }

  return (
    <main
      className="relative bg-[#0a0c15]"
      style={{ paddingLeft: `${SIDEBAR_W}px` }}
    >
      <style>{`
        /* Desktop: full height, no scroll, vertically centered */
        @media (min-width: 768px) {
          main {
            height: 100dvh;
            overflow: hidden;
            display: flex;
            align-items: center;
            justify-content: center;
          }
        }
        /* Mobile: scrollable, topbar offset, sidebar removed */
        @media (max-width: 767px) {
          main {
            padding-left: 0 !important;
            min-height: 100dvh;
            display: flex;
            align-items: center;
            justify-content: center;
            overflow-y: auto;
          }
        }
      `}</style>

      <div className="relative z-10 w-full max-w-4xl lg:max-w-6xl xl:max-w-7xl px-5 sm:px-8 lg:px-12 flex flex-col gap-5 sm:gap-6 lg:gap-7 py-[calc(56px+2rem)] md:py-10 lg:py-12">

            {home.hero.subtitle}
          </p>
